% Replace <<PLACEHOLDERS>>. The cover trick: set the cover fill to the SAME color

% has a transparent/white background instead, drop \includegraphics onto the
% brand fill directly and skip the color-match.

\documentclass[11pt,oneside]{book}
\input{preamble}

\begin{document}

\begin{titlepage}
\thispagestyle{empty}
\begin{tikzpicture}[remember picture, overlay]
  \fill[brandmain] (current page.north west) rectangle (current page.south east);
  % faint geometric accents, top-right
  \begin{scope}
    \clip (current page.north west) rectangle (current page.south east);
    \draw[line width=16mm, white, opacity=0.06]
      ([shift={(-1cm,-1cm)}]current page.north east) circle (7cm);
    \fill[brandaccent, opacity=0.10]
      ([shift={(-3.2cm,-3.2cm)}]current page.north east) circle (2cm);
  \end{scope}

  % logo monogram
  \node[anchor=north west] at ([shift={(2.5cm,-3cm)}]current page.north west)
    {\includegraphics[width=2.6cm]{figures/<<LOGO_FILE>>}};
  % wordmark
  \node[anchor=west, text=white, font=\displayfont\Huge\bfseries]
    at ([shift={(5.6cm,-4.3cm)}]current page.north west) {<<PRODUCT>>};

  % title block
  \node[anchor=west, text=white, font=\displayfont\fontsize{46}{50}\selectfont\bfseries]
    at ([shift={(2.5cm,-11cm)}]current page.north west) {<<TITLE>>};
  \node[anchor=west, text=brandmainlt, font=\Large]
    at ([shift={(2.6cm,-12.6cm)}]current page.north west) {<<SUBTITLE>>};
  \fill[brandaccent] ([shift={(2.6cm,-13.5cm)}]current page.north west)
    rectangle ([shift={(8.6cm,-13.7cm)}]current page.north west);

  % footer meta
  \node[anchor=west, text=white, font=\bfseries]
    at ([shift={(2.5cm,3.1cm)}]current page.south west) {<<DOC_KIND>>};
  \node[anchor=west, text=brandmainlt, font=\small]
    at ([shift={(2.5cm,2.4cm)}]current page.south west) {<<FOOTER_META>> \quad\textbullet\quad \today};
  \fill[brandink] (current page.south west) rectangle ([yshift=6mm]current page.south east);
\end{tikzpicture}
\end{titlepage}

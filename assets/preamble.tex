% --- Subtle branded page background --------------------------------------
\newcommand{\brandpagebg}{%
  \AddToShipoutPictureBG{%
    \begin{tikzpicture}[remember picture, overlay]
      \fill[brandmain] (current page.north west) rectangle
        ([xshift=3.2mm]current page.south west);
      \fill[brandaccent] ([xshift=3.2mm]current page.north west) rectangle
        ([xshift=4.0mm]current page.south west);
      \begin{scope}
        \clip (current page.south east) rectangle ([shift={(-7cm,7cm)}]current page.south east);
        \draw[line width=10mm, brandmain, opacity=0.05]
          ([shift={(-1cm,1cm)}]current page.south east) circle (4.2cm);
        \fill[brandalt, opacity=0.045]
          ([shift={(-1cm,1cm)}]current page.south east) circle (1.7cm);
      \end{scope}
    \end{tikzpicture}}%
}
